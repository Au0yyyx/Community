local function GetClosestPad()
    local localChar = getChar(plr)
    if not localChar then return nil end

    local root,hitbox = getRoot(localChar)
    if not root then return nil end

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {localChar}

    local badColor = Color3.fromRGB(152, 24, 24)
    local closest = nil
    local dist = 100

    for _, part in pads:GetChildren() do
        if part.Color == badColor then print("bad color (NULL GUI)") continue end
        local mag = (root.Position - part.Position).Magnitude
        if mag > dist then print(mag, ">", dist, "part too far (NULL GUI)") continue end

        local origin = root.Position
        local direction = part.Position - origin

        local result = workspace:Raycast(origin, direction, rayParams)
        local visible = false

        if result then
            local hit = result.Instance
            if hit:IsDescendantOf(pads) then
                visible = true
            else
                visible = false
            end
        end

        if visible then
            dist = mag
            closest = part
        end
    end

    return closest
end

local function getSeamines()
    local seamines = {}

    for _, sm in pads:GetChildren() do

            table.insert(seamines, sm)
        end
    end

    return seamines
end

local function isdirectionsafetopushfromguardianbullet(pos, direction)
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    rayParams.FilterDescendantsInstances = {game.Players.LocalPlayer.Character}

    local result = workspace:Raycast(pos, direction * 6, rayParams)
    return result ~= nil and result.Instance and result.Instance.CanCollide == true
end

local function collect(which, suppressNotFound)
    local activeTripmines = getActiveTripmines()
    if magSlider then
        magSlider:Set(5)
    end

    local function collectGolden()
        if tweening then
            if notifOn then
                notif("Already collecting.", "Collection System")
            end
            return
        end
        tweening = true
        local startRefreshing
        local tween:Tween

        while tweening do
            if suppressNotFound then
                local inRound = ReplicatedStorage:FindFirstChild("InRound")
                if not inRound or not inRound.Value then
                    break
                end
            end

            local char = getChar(plr)
            local root = getRoot(char)
            if root then root.AssemblyLinearVelocity = Vector3.new(0,0,0) end

            if not startRefreshing then
                refreshGifts(true, true)
            elseif tick() - startRefreshing >= 15 then
                startRefreshing = tick()
                refreshGifts(true, true)
            end

            local gift = getClosestGift(availableGoldenGifts)
            if not gift then
                if notifOn and not suppressNotFound then
                    notif("no golden gifts, or currently finding golden gifts", "Gift Not Found")
                end
                break
            end
            if not startRefreshing then
                startRefreshing = tick()
            end

            tween = goTo(gift, activeTripmines, enemies:GetChildren())
            if tween then tween.Completed:Wait() end

            task.wait(.01)
        end
        if tween then
            tween:Cancel()
        end
        tweening = false
    end

    local function collectNormal(getGoldenAfter)
        if tweening then
            if notifOn then
                notif("Already collecting.", "Collection System")
            end
            return
        end
        tweening = true
        local startRefreshing
        local tween:Tween

        while tweening do
            if suppressNotFound then
                local inRound = ReplicatedStorage:FindFirstChild("InRound")
                if not inRound or not inRound.Value then
                    break
                end
            end

            local char = getChar(plr)
            local root = getRoot(char)
            if root then root.AssemblyLinearVelocity = Vector3.new(0,0,0) end

            if not startRefreshing then
                refreshGifts(true, false)
            elseif tick() - startRefreshing >= 15 then
                startRefreshing = tick()
                refreshGifts(true, false)
            end

            local gift = getClosestGift(availableNormalGifts)
            if not gift then
                if notifOn and not suppressNotFound then
                    notif("no gifts, or currently finding gifts", "Gift Not Found")
                end
                break
            end
            if not startRefreshing then
                startRefreshing = tick()
            end

            tween = goTo(gift, activeTripmines, enemies:GetChildren())
            if tween then tween.Completed:Wait() end

            task.wait(.01)
        end
        if tween then
            tween:Cancel()
        end
        tweening = false
        if getGoldenAfter then task.wait(3) collectGolden() end
    end

    if which == "normal" then
        collectNormal()
    elseif which == "golden" then
        collectGolden()
    end
end

    function controller:CompactCandidates(candidateList)
        local seenGifts = {}

        for candidateIndex = #candidateList, 1, -1 do
            local gift = candidateList[candidateIndex]
            local validGift = gift
                and gift.Parent
                and gift.Transparency ~= 1
                and gift:FindFirstChild("Collect", true)

            if not validGift or seenGifts[gift] then
                table.remove(candidateList, candidateIndex)
            else
                seenGifts[gift] = true
            end
        end
    end

        local beacon = workspace:FindFirstChild("Beacon")
        local character = getChar(plr)
        local root, hitbox = getRoot(character)
        if not beacon or not root then
            return false
        end

        local beaconPosition = beacon:IsA("Model") and beacon:GetPivot().Position or beacon.Position
        local destination = CFrame.new(beaconPosition + Vector3.new(0, 4, 0))
        root.CFrame = destination

        if hitbox then
            hitbox.CFrame = destination
        end

        return true
    end

        return option:IsA("BasePart") and option.CFrame or option:GetPivot()
    end

    function controller:GetSortedLobbyOptions(selectFolder)
        local options = {}

        for _, option in ipairs(selectFolder:GetChildren()) do
            local optionName = string.lower(option.Name)
            local isRerollOption = optionName:find("reroll", 1, true) ~= nil

            if not isRerollOption and option:FindFirstChildOfClass("ProximityPrompt", true) then
                table.insert(options, option)
            end
        end

        table.sort(options, function(leftOption, rightOption)
            local leftCFrame = self:GetLobbyOptionCFrame(leftOption)
            local rightCFrame = self:GetLobbyOptionCFrame(rightOption)
            local leftOffset = spawnPart.CFrame:ToObjectSpace(leftCFrame).Position.X
            local rightOffset = spawnPart.CFrame:ToObjectSpace(rightCFrame).Position.X
            return leftOffset < rightOffset
        end)

        return options
    end

    function controller:ActivateLobbyOption(option, root)
        local prompt = option and option:FindFirstChildOfClass("ProximityPrompt", true)
        if not prompt or type(fireproximityprompt) ~= "function" then
            return false
        end

        root.CFrame = self:GetLobbyOptionCFrame(option) + Vector3.new(0, 3, 0)
        task.wait(0.3)
        fireproximityprompt(prompt)
        task.wait(0.7)
        return true
    end

    function controller:RunLobbySelection()
        if self.LobbyBusy or self.Destroyed then
            return
        end

        self.LobbyBusy = true
        self.LastLobbyAttempt = tick()

        local selectionSucceeded, selectionError = pcall(function()
            local selectFolder = workspace:FindFirstChild("Select")
            local character = getChar(plr)
            local root = getRoot(character)
            if not selectFolder or not spawnPart or not root then
                return
            end

            local options = self:GetSortedLobbyOptions(selectFolder)
            if #options == 0 then
                return
            end

            self:ActivateLobbyOption(options[#options], root)
            options = self:GetSortedLobbyOptions(selectFolder)

            table.sort(options, function(leftOption, rightOption)
                local leftOffset = spawnPart.CFrame:ToObjectSpace(self:GetLobbyOptionCFrame(leftOption)).Position.X
                local rightOffset = spawnPart.CFrame:ToObjectSpace(self:GetLobbyOptionCFrame(rightOption)).Position.X
                return math.abs(leftOffset) < math.abs(rightOffset)
            end)

            if options[1] then
                self:ActivateLobbyOption(options[1], root)
            end

            task.wait(1.3)
        end)

        self.LobbyBusy = false
        if not selectionSucceeded then
            warn("[NULLGUI Auto-Farm] Lobby selection failed: " .. tostring(selectionError))
        end
    end

    function controller:StartCollection(collectionType)
        if self.CollectionTaskActive or tweening or self.Destroyed then

                and self.AutoBeacon
                and self.SawGoldenGift
                and not self.BeaconedForCurrentBatch
                and not getClosestGift(availableGoldenGifts)
            then
                self.BeaconedForCurrentBatch = self:TeleportToBeacon()
            end
        end)

        return true
    end

        self.AutoGolden = false

                end
            end
        end
    end

    function controller:Destroy()

        end

        self.Destroyed = true

    local intv:IntValue = upgrades:FindFirstChild(name)
    if intv and intv.Value > 1 then
        print("intv found, more than 1")
        intv.Value -= 1
        fireSig(events.UpgradesChanged.OnClientEvent, {
            [name] = intv.Value
        })

        return
    elseif intv and intv.Value <= 1 then
        print("intv found, 1, destroying...")
        intv:Destroy()
        fireSig(events.UpgradesChanged.OnClientEvent, {
            [name] = 0
        })
        return
    end

    fireSig(events.UpgradesChanged.OnClientEvent, {
        [name] = 0
    })
    print("no int found, resetting to 0")
end

            return 0
        end

        return math.clamp(math.floor(numericValue), 0, 999)
    end

    function controller:SetUpgradeQuantity(upgradeName, quantity)
        local normalizedQuantity = self:NormalizeQuantity(quantity)
        if normalizedQuantity <= 0 or self.Destroyed then
            return false
        end

        local upgradeValue = upgrades:FindFirstChild(upgradeName)
        if not upgradeValue then
            upgradeValue = Instance.new("IntValue")
            upgradeValue.Name = upgradeName
            upgradeValue.Parent = upgrades

            local character = plr.Character
            local humanoid = character and getHuman(character)
            local root = humanoid and humanoid.RootPart
            if root then
                root.Destroying:Once(function()
                    if upgradeValue.Parent then
                        upgradeValue:Destroy()
                    end
                end)
            end
        end

        upgradeValue.Value = normalizedQuantity
        fireSig(events.UpgradesChanged.OnClientEvent, {
            [upgradeName] = normalizedQuantity
        })
        return true
    end

    function controller:ApplyAll()
        if self.Applying or self.Destroyed then
            return false
        end

        if not fSignal then
            if notifOn then
                notif("Your executor cannot apply client upgrades.", "Auto-Add Upgrades")
            end
            return false
        end

        self.Applying = true
        local appliedCount = 0

        for upgradeIndex, upgradeName in ipairs(clientUpgrades) do
            local quantity = self.Quantities[upgradeName] or 0
            if quantity > 0 and self:SetUpgradeQuantity(upgradeName, quantity) then
                appliedCount += 1
            end

            if upgradeIndex % 5 == 0 then
                task.wait()
            end
        end

        self.Applying = false

        if appliedCount > 0 and notifOn then
            notif("Re-added " .. tostring(appliedCount) .. " configured upgrade(s).", "Auto-Add Upgrades")
        end

        return appliedCount > 0
    end

    function controller:ScheduleApply(delaySeconds)
        task.delay(delaySeconds or 0, function()
            if not self.Destroyed and not destroying then
                self:ApplyAll()
            end
        end)
    end

    function controller:Destroy()
        self.Destroyed = true
        self.Applying = false
    end

    for _, upgradeName in ipairs(clientUpgrades) do
        upgradeTab:CreateSection(upgradeName)

        controller.Quantities[upgradeName] = 0

        upgradeTab:CreateLabel("Auto-add this upgrade!")
        upgradeTab:CreateInput({
            Name = "Auto-Add Quantity",
            CurrentValue = "0",
            PlaceholderText = "Example: 30",
            RemoveTextAfterFocusLost = false,
            Flag = "NULLGUIAutoAddUpgrade_" .. upgradeName,
            Callback = function(value)
                controller.Quantities[upgradeName] = controller:NormalizeQuantity(value)
            end
        })
        upgradeTab:CreateButton({
            Name = "Remove One",
            Callback = function()
                subUpgrade(upgradeName)
            end
        })
        upgradeTab:CreateButton({
            Name = "Add One",
            Callback = function()
                addUpgrade(upgradeName)
            end
        })
    end

    table.insert(connections, ReplicatedStorage.InRound.Changed:Connect(function(roundIsActive)
        controller:ScheduleApply(roundIsActive and 0.25 or 1.25)
    end))

local function disableAll(willDestroy: boolean, client: boolean, willBreakAI: boolean)
    willDestroy = if willDestroy == nil then false else willDestroy
    client = if client == nil then false else client
    willBreakAI = if willBreakAI == nil then false else willBreakAI

    local allenemies = enemies:GetChildren()
    if not allenemies or #allenemies == 0 then
        if notifOn then
            notif("No enemies available.", "Not found")
        end
        return
    end

    if client then
        for _, enemy in clientenemies do
            if not enemies:FindFirstChild(enemy) then continue end
            disableEnemy(enemy, willDestroy, willBreakAI)
        end

        return
    end

    for _, enemy in allenemies do
        disableEnemy(enemy, willDestroy, willBreakAI)
    end
end
