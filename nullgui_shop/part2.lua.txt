    if not char then return nil end

    return char:FindFirstChild("HumanoidRootPart") or (humanoid and humanoid.RootPart), char:FindFirstChild("Hitbox")

local function isDead(target: Player? | Model?)
    if target:IsA("Player") then
        local char = target.Character
        local humanoid: Humanoid = char and getHuman(char)
        return char and (humanoid and (humanoid:GetState() == Enum.HumanoidStateType.Dead or humanoid.Health <= 0))
            or char == nil
    elseif target:IsA("Model") then
        local humanoid = target and getHuman(target)
        return humanoid and (humanoid:GetState() == Enum.HumanoidStateType.Dead or humanoid.Health <= 0)
            or target == nil
    else
        return target == nil
    end
end

local function isInsideCelestial(instance)
    local currentInstance = instance

    while currentInstance and currentInstance ~= workspace do
        if isCelestialName(currentInstance.Name) then
            local ancestor = currentInstance
            while ancestor do
                if ancestor == enemies then
                    return true
                end
                ancestor = ancestor.Parent
            end
            return false
        end

        currentInstance = currentInstance.Parent
    end

    return false
end

local function shouldBlockCelestialRemote(remote, methodName, arguments)
    if celestialIntegration.Controller.Destroyed then
        return false
    end

    if remote == speakingRemote
        and methodName == "FireServer"
        and typeof(arguments[1]) == "boolean" then
        return true
    end

    local isDeathRemote = remote == diedRemote and methodName == "FireServer"
    local isDeathFunction = remote == diedFunction and methodName == "InvokeServer"

    if (isDeathRemote or isDeathFunction) and typeof(arguments[1]) == "string" then
        local deathCause = string.lower(arguments[1])
        if celestialIntegration.Controller.EnemySafety.Friendly[deathCause] then
            celestialIntegration.Controller.EnemySafety.BlockedDeathReports += 1
            return true
        end
    end

    if not celestialIntegration.Controller.Enabled then
        return false
    end

    if (isDeathRemote or isDeathFunction) and isCelestialDeathCause(arguments[1]) then
        celestialIntegration.Controller.BlockedDeathReports += 1
        return true
    end

    if methodName ~= "FireServer" or not isCelestialClientEvent(remote) then
        return false
    end

    if arguments[1] == "Kill" then
        celestialIntegration.Controller.BlockedClientEvents += 1
        return true
    end

    return false
end

local function trySharkTailCancel(movement)
    local safety = celestialIntegration.Controller.EnemySafety
    safety.SharkTailCancelAttempts += 1

    if type(movement) ~= "table" or type(movement.HasUpgrade) ~= "function" then
        return false
    end

    local checkedUpgrade, hasSharkTail = pcall(movement.HasUpgrade, movement, "SharkTail")
    if not checkedUpgrade or hasSharkTail ~= true then
        return false
    end

    local activeSpecial = movement.ActiveSpecial
    if type(activeSpecial) ~= "table" then
        return false
    end

    local route
    local cancelled
    if movement.SpecialID == "Charger" and type(movement.SpecialAction) == "function" then
        if movement.charging == true and movement.chargeSlowing ~= true then
            local releaseState = movement.toggleAbility == true
                and Enum.UserInputState.Begin or Enum.UserInputState.End
            pcall(movement.SpecialAction, movement, releaseState)
        end
        if movement.chargeSlowing == true then
            route = "ChargerReleaseThenSecondPress"
            cancelled = pcall(
                movement.SpecialAction,
                movement,
                Enum.UserInputState.Begin
            )
        end
    elseif movement.diving == true and type(movement.SpecialAction) == "function" then
        route = "DiveSecondPress"
        cancelled = pcall(movement.SpecialAction, movement, Enum.UserInputState.Begin)
    elseif movement.SpecialID == "Grapple"
        and movement.grappling == true
        and activeSpecial.grappleReel == true
        and type(movement.AltSpecialAction) == "function"
        and type(movement.SpecialAction) == "function" then
        route = "GrappleReelSecondPress"
        local releasedAlt = pcall(
            movement.AltSpecialAction,
            movement,
            Enum.UserInputState.End
        )
        if releasedAlt and activeSpecial.allowSharkTail == true then
            cancelled = pcall(
                movement.SpecialAction,
                movement,
                Enum.UserInputState.Begin
            )
        end
    elseif movement.SpecialID == "Grapple"
        and activeSpecial.allowSharkTail == true
        and activeSpecial.sharkTailDisable ~= true
        and type(movement.SpecialAction) == "function" then
        route = "GrappleSecondPress"
        cancelled = pcall(movement.SpecialAction, movement, Enum.UserInputState.Begin)
    end

    if cancelled then
        safety.SharkTailCancels += 1
        safety.LastSharkTailRoute = route
        return true
    end
    return false
end

            or (activeSpecial and activeSpecial.grappleReel == true)
            or (activeSpecial and activeSpecial.allowSharkTail == true)
            or (humanoid and humanoid:GetAttribute("UsingAbility") == true)
    end)
    return readState and active

            return nil, nil
        end

        return humanoid, rootPart
    end

    function AutoStopController:GetControls()
        if self.Controls then
            return self.Controls
        end

        local getter = self.Dependencies.GetControls
        if type(getter) ~= "function" then
            return nil
        end

        local success, controls = pcall(getter)
        if success and type(controls) == "table" then
            self.Controls = controls
            return controls
        end

        return nil
    end

    function AutoStopController:ReleaseKeys()
        local virtualInputManager = self.Dependencies.VirtualInputManager
        local userInputService = self.Dependencies.UserInputService
        if not virtualInputManager then
            return
        end

        local keysToRelease = {}
        local queuedKeys = {}
        local function queueKey(keyCode)
            if keyCode == Enum.KeyCode.Thumbstick1
                or keyCode == Enum.KeyCode.Thumbstick2 then
                return
            end

            if typeof(keyCode) == "EnumItem"
                and keyCode ~= Enum.KeyCode.Unknown
                and not queuedKeys[keyCode] then
                queuedKeys[keyCode] = true
                table.insert(keysToRelease, keyCode)
            end
        end

        for _, keyCode in ipairs(self.Dependencies.ReleaseKeys or DEFAULT_RELEASE_KEYS) do
            queueKey(keyCode)
        end

        if userInputService then
            local readKeys, pressedKeys = pcall(userInputService.GetKeysPressed, userInputService)
            if readKeys then
                for _, inputObject in ipairs(pressedKeys) do
                    queueKey(inputObject.KeyCode)
                end
            end

            local readGamepads, connectedGamepads = pcall(
                userInputService.GetConnectedGamepads,
                userInputService
            )
            if readGamepads then
                for _, gamepad in ipairs(connectedGamepads) do
                    local readState, gamepadState = pcall(
                        userInputService.GetGamepadState,
                        userInputService,
                        gamepad
                    )
                    if readState then
                        for _, inputObject in ipairs(gamepadState) do
                            if inputObject.UserInputState ~= Enum.UserInputState.End then
                                queueKey(inputObject.KeyCode)
                            end
                        end
                    end
                end
            end
        end

        for _, keyCode in ipairs(keysToRelease) do
            safeCall(function()
                virtualInputManager:SendKeyEvent(false, keyCode, false, game)
            end)
        end

        safeCall(function()
            if not userInputService then
                return
            end

            local mousePosition = userInputService:GetMouseLocation()
            for buttonIndex = 0, 2 do
                virtualInputManager:SendMouseButtonEvent(
                    mousePosition.X,
                    mousePosition.Y,
                    buttonIndex,
                    false,
                    game,
                    0
                )
            end
        end)
    end

    function AutoStopController:ReleaseAbility(cancelMethods)
        self:ReleaseKeys()
        safeCall(self.Dependencies.ReleaseAbility, cancelMethods)
        safeCall(self.Dependencies.ReleaseMovementInput)

        local humanoid = self:GetParts()
        if humanoid then
            safeCall(function()
                humanoid:SetAttribute("UsingAbility", false)
                humanoid:SetAttribute("Sliding", false)
                humanoid:SetAttribute("PlayerMoveDirection", Vector3.zero)
            end)
        end
    end

    function AutoStopController:Capture()
        if self.Snapshot then
            return
        end

        local controls = self:GetControls()
        self.Snapshot = {
            Controls = controls,
            ControlsEnabled = readProperty(controls, "controlsEnabled"),
        }
    end

    function AutoStopController:Apply(humanoid)
        self:Capture()
        if not self.AbilityReleased then
            self:ReleaseAbility()
            self.AbilityReleased = true
        end

        local controls = self:GetControls()
        safeCall(function()
            controls:Disable()
        end)
        safeCall(self.Dependencies.ReleaseMovementInput)

        humanoid = humanoid or self:GetParts()
        if humanoid then
            safeCall(function()
                humanoid:SetAttribute("PlayerMoveDirection", Vector3.zero)
                humanoid:SetAttribute("UsingAbility", false)
                humanoid:SetAttribute("Sliding", false)
            end)
        end

        return true
    end

    function AutoStopController:Restore()
        local snapshot = self.Snapshot
        self.Snapshot = nil
        self.Active = false
        self.AbilityReleased = false

        local runService = self.Dependencies.RunService
        if runService then
            runService:UnbindFromRenderStep(self.InputNeutralBindingName)
        end

        if not snapshot then
            self:ReleaseKeys()
            return
        end

        if snapshot.Controls then
            if snapshot.ControlsEnabled == false then
                safeCall(function()
                    snapshot.Controls:Disable()
                end)
            else
                safeCall(function()
                    snapshot.Controls:Enable()
                end)
            end
        end

        self:ReleaseKeys()
        safeCall(self.Dependencies.OnRestored)
    end

    function AutoStopController:Finish()
        self.Generation += 1
        self.Deadline = 0

        if self.HeartbeatConnection then
            safeCall(function()
                self.HeartbeatConnection:Disconnect()
            end)
            self.HeartbeatConnection = nil
        end

        self:Restore()
    end

    function AutoStopController:Cancel()
        self.ScheduleGeneration += 1
        self:Finish()
    end

    function AutoStopController:StopFor(duration, abilityAlreadyReleased)
        duration = math.max(tonumber(duration) or 0, 0)
        self.Generation += 1
        local generation = self.Generation
        self.Deadline = os.clock() + duration
        self.Active = true
        self.AbilityReleased = abilityAlreadyReleased == true

        local runService = self.Dependencies.RunService
        if runService then
            runService:UnbindFromRenderStep(self.InputNeutralBindingName)
            runService:BindToRenderStep(
                self.InputNeutralBindingName,
                Enum.RenderPriority.Last.Value + 100,
                function()
                    if not self.Active then
                        return
                    end

                    safeCall(self.Dependencies.ReleaseMovementInput)
                    local currentHumanoid = self:GetParts()
                    self:Apply(currentHumanoid)
                end
            )
        end

        if self.HeartbeatConnection then
            safeCall(function()
                self.HeartbeatConnection:Disconnect()
            end)
            self.HeartbeatConnection = nil
        end

        local humanoid = self:GetParts()
        self:Apply(humanoid)
        safeCall(self.Dependencies.OnStopped)

        if runService and duration > 0 then
            self.HeartbeatConnection = runService.Heartbeat:Connect(function()
                if generation ~= self.Generation or not self.Active then
                    return
                end

                if os.clock() >= self.Deadline then
                    self:Finish()
                    return
                end

                local currentHumanoid = self:GetParts()
                self:Apply(currentHumanoid)
            end)
        elseif duration <= 0 then
            self:Finish()
        end
    end

    function AutoStopController:WaitForAbilityRelease(scheduleGeneration)
        local runService = self.Dependencies.RunService
        if not runService then
            return scheduleGeneration == self.ScheduleGeneration
        end

        local deadline = os.clock() + abilityReleaseSettleTimeout
        repeat
            runService.Heartbeat:Wait()

            if scheduleGeneration ~= self.ScheduleGeneration
                or celestialIntegration.Controller.Destroyed then
                return false
            end

            safeCall(self.Dependencies.ReleaseMovementInput)
            local humanoid = self:GetParts()
            if not humanoid or humanoid:GetAttribute("UsingAbility") ~= true then
                return true
            end
        until os.clock() >= deadline

        return true
    end

    function AutoStopController:StopAfter(
        delay,
        duration,
        activation,
        getActivation,
        cancelMethods
    )
        delay = math.max(tonumber(delay) or 0, 0)
        duration = math.max(tonumber(duration) or 0, 0)
        local scheduleGeneration = self.ScheduleGeneration

        task.delay(delay, function()
            if scheduleGeneration ~= self.ScheduleGeneration
                or celestialIntegration.Controller.Destroyed then
                return
            end

            if type(getActivation) == "function" and activation ~= nil then
                local success, currentActivation = pcall(getActivation)
                if not success or currentActivation ~= activation then
                    return
                end
            end

            self:ReleaseAbility(cancelMethods)
            if not self:WaitForAbilityRelease(scheduleGeneration) then
                return
            end

            self:StopFor(duration, true)
        end)
    end

    function AutoStopController:Destroy()
        self:Cancel()
    end

    return AutoStopController.new({
        GetParts = getCharacterControlParts,
        GetControls = function()
            local playerScripts = plr:FindFirstChild("PlayerScripts")
            local playerModuleScript = playerScripts and playerScripts:FindFirstChild("PlayerModule")
            if not playerModuleScript then
                return nil
            end

            local loaded, playerModule = pcall(require, playerModuleScript)
            if not loaded or type(playerModule) ~= "table"
                or type(playerModule.GetControls) ~= "function" then
                return nil
            end

            return playerModule:GetControls()
        end,
        VirtualInputManager = VirtualInputManager,
        UserInputService = UserInputService,
        RunService = RunService,
        ReleaseMovementInput = function()
            local movement = getAbilityController()
            local zero = Vector3.zero
            local controlModule = movement and movement.ControlModule
            if type(controlModule) ~= "table" then
                local playerScripts = plr:FindFirstChild("PlayerScripts")
                local playerModuleScript = playerScripts and playerScripts:FindFirstChild("PlayerModule")
                if playerModuleScript then
                    local loaded, playerModule = pcall(require, playerModuleScript)
                    if loaded and type(playerModule) == "table"
                        and type(playerModule.GetControls) == "function" then
                        local foundControls, controls = pcall(playerModule.GetControls, playerModule)
                        if foundControls then
                            controlModule = controls
                        end
                    end
                end
            end

            if type(controlModule) == "table" then
                controlModule.inputMoveVector = zero

                local activeController
                if type(controlModule.GetActiveController) == "function" then
                    local foundController, controller = pcall(
                        controlModule.GetActiveController,
                        controlModule
                    )
                    if foundController then
                        activeController = controller
                    end
                else
                    activeController = controlModule.activeController
                end

                if type(activeController) == "table" then
                    activeController.moveVector = zero
                    activeController.forwardValue = 0
                    activeController.backwardValue = 0
                    activeController.leftValue = 0
                    activeController.rightValue = 0
                    activeController.isJumping = false
                end

                if type(controlModule.moveFunction) == "function" then
                    pcall(controlModule.moveFunction, plr, zero, true)
                end
            end

            if movement then
                movement.specialHeld = false
                movement.altSpecialHeld = false
                movement.jumpPressed = false
                movement.sprintHeld = false
                movement.PlayerMoveDirection = zero
            end

            local humanoid = movement and movement.Humanoid or getCharacterControlParts()
            if humanoid and humanoid.Parent then
                humanoid:SetAttribute("PlayerMoveDirection", zero)
                humanoid:SetAttribute("UsingAbility", false)
                humanoid:SetAttribute("Sliding", false)
            end
        end,
        ReleaseAbility = releaseAbility,
        OnStopped = function()
            celestialIntegration.Controller.EnemySafety.StopsPerformed += 1
        end,
    })
end)()
