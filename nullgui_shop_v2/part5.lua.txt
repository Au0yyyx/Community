local function rememberUpgradeOriginal(name)
    if upgradeOriginals[name] == nil then
        local value = upgrades:FindFirstChild(name)
        upgradeOriginals[name] = value and tonumber(value.Value) or 0
    end
end

local function addUpgrade(name)
    rememberUpgradeOriginal(name)
    local intv:IntValue = upgrades:FindFirstChild(name)
    if intv then
        print("intv found")
        intv.Value += 1
        fireSig(events.UpgradesChanged.OnClientEvent, {
            [name] = intv.Value
        })
    else
        print("no intv")
        fireSig(events.UpgradesChanged.OnClientEvent, {
            [name] = 1
        })

        intv = Instance.new("IntValue")
        intv.Value = 1
        intv.Name = name
        intv.Parent = upgrades

        getHuman(getChar(plr)).RootPart.Destroying:Once(function()
            print("destroyed", intv)
            intv:Destroy()
        end)
    end

end
local function subUpgrade(name)
    rememberUpgradeOriginal(name)
    local intv:IntValue = upgrades:FindFirstChild(name)
    if intv and intv.Value > 1 then
        print("intv found, more than 1")
        intv.Value -= 1
        fireSig(events.UpgradesChanged.OnClientEvent, {
            [name] = intv.Value
        })

        return
    elseif intv and intv.Value <= 1 then
        print("intv found, 1, destroying...")
        intv:Destroy()
        fireSig(events.UpgradesChanged.OnClientEvent, {
            [name] = 0
        })
        return
    end

    fireSig(events.UpgradesChanged.OnClientEvent, {
        [name] = 0
    })
    print("no int found, resetting to 0")
end

(function()
    local controller = {
        Destroyed = false,
        Applying = false,
        Quantities = {},
    }

    celestialIntegration.UpgradeAutomation = controller

    function controller:NormalizeQuantity(value)
        local numericValue = tonumber(value)
        if not numericValue then
            return 0
        end

        return math.clamp(math.floor(numericValue), 0, 999)
    end

    function controller:SetUpgradeQuantity(upgradeName, quantity)
        rememberUpgradeOriginal(upgradeName)
        local normalizedQuantity = self:NormalizeQuantity(quantity)
        if normalizedQuantity <= 0 or self.Destroyed then
            return false
        end

        local upgradeValue = upgrades:FindFirstChild(upgradeName)
        if not upgradeValue then
            upgradeValue = Instance.new("IntValue")
            upgradeValue.Name = upgradeName
            upgradeValue.Parent = upgrades

            local character = plr.Character
            local humanoid = character and getHuman(character)
            local root = humanoid and humanoid.RootPart
            if root then
                root.Destroying:Once(function()
                    if upgradeValue.Parent then
                        upgradeValue:Destroy()
                    end
                end)
            end
        end

        upgradeValue.Value = normalizedQuantity
        fireSig(events.UpgradesChanged.OnClientEvent, {
            [upgradeName] = normalizedQuantity
        })
        return true
    end

    function controller:ApplyAll()
        if self.Applying or self.Destroyed then
            return false
        end

        if not fSignal then
            if notifOn then
                notif("Your executor cannot apply client upgrades.", "Auto-Add Upgrades")
            end
            return false
        end

        self.Applying = true
        local appliedCount = 0

        for upgradeIndex, upgradeName in ipairs(clientUpgrades) do
            local quantity = self.Quantities[upgradeName] or 0
            if quantity > 0 and self:SetUpgradeQuantity(upgradeName, quantity) then
                appliedCount += 1
            end

            if upgradeIndex % 5 == 0 then
                task.wait()
            end
        end

        self.Applying = false

        if appliedCount > 0 and notifOn then
            notif("Re-added " .. tostring(appliedCount) .. " configured upgrade(s).", "Auto-Add Upgrades")
        end

        return appliedCount > 0
    end

    function controller:ScheduleApply(delaySeconds)
        task.delay(delaySeconds or 0, function()
            if not self.Destroyed and not destroying then
                self:ApplyAll()
            end
        end)
    end

    function controller:Destroy()
        self.Destroyed = true
        self.Applying = false
    end

local function disableAll(willDestroy: boolean, client: boolean, willBreakAI: boolean)

        if notifOn then

            if not enemies:FindFirstChild(enemy) then continue end
            disableEnemy(enemy, willDestroy, willBreakAI)
        end

        return
    end

    for _, enemy in allenemies do
        disableEnemy(enemy, willDestroy, willBreakAI)
    end
end

local function handleEnemy(enemy)
    local name = enemy.Name
    local waitingTime = 25

    if name == "ICBM" or name == "Telefragger" or name:find("Baby") then
        waitingTime = 75 --lowered to 75
    end

    if auto_destroy[name] then
        local start = tick()
        local didDestroy

        repeat
            if tick() - start >= waitingTime then break end
            didDestroy = disableEnemy(name, true, false, false)
            task.wait(.2)
        until didDestroy == true
    elseif auto_break[name] then
        if name == "Mart" and curses:FindFirstChild("MartSlide") and notifOn then
            notif("Destroy Mart instead of breaking.", "MART SLIDE DETECTED")
        end

        local start = tick()
        local didBreak

        repeat
            if tick() - start >= waitingTime then break end
            didBreak = disableEnemy(name, false, true, false)
            task.wait(.2)
        until didBreak == true
    elseif auto_disable[name] then
        if name == "RealityBreak" then
            celestialIntegration.Controller.EnemySafety.AutoDisableRealityBreak = false
            celestialIntegration.Controller.EnemySafety.Friendly.realitybreak = true
            celestialIntegration.Controller:RefreshSafetyFilter()
            return
        end

        if name == "Mart" and curses:FindFirstChild("MartSlide") and notifOn then
            notif("Destroy Mart instead of disabling.", "MART SLIDE DETECTED")
        end

        local start = tick()
        local didDisable

        repeat
            if tick() - start >= waitingTime then break end
            didDisable = disableEnemy(name, false, false, false)
            task.wait(.2)
        until didDisable == true
    end

end
