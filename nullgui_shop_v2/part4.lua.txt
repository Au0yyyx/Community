local function refreshGifts(skip, golden)
    local currentAvailableGifts = #availableNormalGifts
    if #availableNormalGifts == 0 or golden then
        currentAvailableGifts = #availableGoldenGifts
    end
    local REFRESH_RATE = (currentAvailableGifts > 5000 and 1/0.25) or (currentAvailableGifts > 3000 and 1/1) or (currentAvailableGifts > 1500 and 1/3) or (currentAvailableGifts > 1000 and 1/5) or (currentAvailableGifts > 500 and 1/12.5) or 1/25

    if not skip then
        if tick() - lastRefresh < REFRESH_RATE then return end
        lastRefresh = tick()
    end

    local char = getChar(plr)
    local root = getRoot(char)
    if not root then return end

    local rootPos = root.Position

    for i = 1, SCAN_SIZE do
        local gift = normalList[scanIndex]
        if not gift then
            scanIndex = 1
            break
        end

        if gift.Transparency ~= 1 and gift:FindFirstChild("Collect", true) then
            local dist = (rootPos - gift.Position).Magnitude

            if dist <= 500 then
                availableNormalGifts[#availableNormalGifts+1] = gift
            end
        end

        scanIndex += 1
    end

    for i = 1, SCAN_SIZE do
        local gift = goldenList[scanIndexTwo]
        if not gift then
            scanIndexTwo = 1
            break
        end

        if gift.Transparency ~= 1 and gift:FindFirstChild("Collect", true) then
            local dist = (rootPos - gift.Position).Magnitude

            if dist <= 500 then
                availableGoldenGifts[#availableGoldenGifts+1] = gift
            end
        end

        scanIndexTwo += 1
    end
end

local function pathBlocked(targetPos, activeTripmines, activeEnemies)
    local char = getChar(plr)
    local root = getRoot(char)
    if not root then return true end

    local rootPos = root.Position
    local fakeSize = Vector3.new(2,5,2)
    
    local minX = math.min(rootPos.X - fakeSize.X/2, targetPos.X - fakeSize.X/2)
    local minY = math.min(rootPos.Y - fakeSize.Y/2, targetPos.Y - fakeSize.Y/2)
    local minZ = math.min(rootPos.Z - fakeSize.Z/2, targetPos.Z - fakeSize.Z/2)
    local maxX = math.max(rootPos.X + fakeSize.X/2, targetPos.X + fakeSize.X/2)
    local maxY = math.max(rootPos.Y + fakeSize.Y/2, targetPos.Y + fakeSize.Y/2)
    local maxZ = math.max(rootPos.Z + fakeSize.Z/2, targetPos.Z + fakeSize.Z/2)

    for _, mine in activeTripmines do
        local pos = mine.Position
        local size = mine.Size
        local minMx, maxMx = pos.X - size.X/2, pos.X + size.X/2
        local minMy, maxMy = pos.Y - size.Y/2, pos.Y + size.Y/2
        local minMz, maxMz = pos.Z - size.Z/2, pos.Z + size.Z/2

        local overlapX = maxX >= minMx and minX <= maxMx
        local overlapY = maxY >= minMy and minY <= maxMy
        local overlapZ = maxZ >= minMz and minZ <= maxMz

        if overlapX and overlapY and overlapZ then
            return true
        end
    end
    for _, enemy in activeEnemies do
        local pos = enemy.Position
        local size = enemy.Size
        local minMx, maxMx = pos.X - size.X/2, pos.X + size.X/2
        local minMy, maxMy = pos.Y - size.Y/2, pos.Y + size.Y/2
        local minMz, maxMz = pos.Z - size.Z/2, pos.Z + size.Z/2

        local overlapX = maxX >= minMx and minX <= maxMx
        local overlapY = maxY >= minMy and minY <= maxMy
        local overlapZ = maxZ >= minMz and minZ <= maxMz

        if overlapX and overlapY and overlapZ then
            return true
        end
    end

    return false
end

local function goTo(part, activeTripmines, activeEnemies)
    if not activeEnemies then activeEnemies = enemies:GetChildren() end
    if not part then return end
    local char = getChar(plr)
    local root, hitbox = getRoot(char)
    local rootPos = root and root.Position
    if not root or not hitbox then return end

    local pos = part:IsA("Model") and part:GetPivot().Position or part.Position
    if part.Name == "Spawn" then pos += Vector3.new(0,4,0) end
    local diff = pos - rootPos
    local dist = diff.Magnitude
    if dist == 0 or dist >= 10000 then return end

    local direction = diff.Unit
    root.CFrame = CFrame.new(root.Position, root.Position + direction) * CFrame.Angles(0, math.rad(90), 0)

    local blocked = pathBlocked(pos, activeTripmines, activeEnemies)
    if blocked and (part.Name == "Gift" or part.Name == "GoldenGift") then
        root.Position = pos
        task.wait(.3)
        return
    end

    local info = TweenInfo.new(dist / 120, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut)
    local tween = TweenService:Create(root, info, {Position = pos})
    tween:Play()
    TweenService:Create(hitbox, info, {Position = pos}):Play()

    task.spawn(function()
        while tween.PlaybackState == Enum.PlaybackState.Playing do
            if pathBlocked(pos, activeTripmines, activeEnemies) then
                tween:Cancel()
                break
            end
            task.wait(0.05)
        end
    end)
    tween.Completed:Once(function()
        hitbox.Position = root.Position
    end)

    return tween
end

                continue
            end

            local loaded = sameenemy:FindFirstChild(remove, true) or sameenemy:FindFirstChild(name.."_ClientAI", true) or sameenemy:GetAttribute("Disabled") == true

            end

            total += 1

                    disabledThisEnemy = true
                end
            end

                sameenemy:SetAttribute("Disabled", true)
                n += 1
            end
        end

        return n, total, enemiesFound

local function GetClosestPad()
    local localChar = getChar(plr)
    if not localChar then return nil end

    local root,hitbox = getRoot(localChar)
    if not root then return nil end

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {localChar}

    local badColor = Color3.fromRGB(152, 24, 24)
    local closest = nil
    local dist = 100

    for _, part in pads:GetChildren() do
        if part.Color == badColor then print("bad color (NULL GUI)") continue end
        local mag = (root.Position - part.Position).Magnitude
        if mag > dist then print(mag, ">", dist, "part too far (NULL GUI)") continue end

        local origin = root.Position
        local direction = part.Position - origin

        local result = workspace:Raycast(origin, direction, rayParams)
        local visible = false

        if result then
            local hit = result.Instance
            if hit:IsDescendantOf(pads) then
                visible = true
            else
                visible = false
            end
        end

        if visible then
            dist = mag
            closest = part
        end
    end

    return closest
end

            if not startRefreshing then
                refreshGifts(true, false)
            elseif tick() - startRefreshing >= 15 then
                startRefreshing = tick()
                refreshGifts(true, false)
            end

            local gift = getClosestGift(availableNormalGifts)
            if not gift then
                if notifOn and not suppressNotFound then
                    notif("no gifts, or currently finding gifts", "Gift Not Found")
                end
                break
            end
            if not startRefreshing then
                startRefreshing = tick()
            end

            tween = goTo(gift, activeTripmines, enemies:GetChildren())
            if tween then tween.Completed:Wait() end

            task.wait(.01)
        end
        if tween then
            tween:Cancel()
        end
        tweening = false
        if getGoldenAfter then task.wait(3) collectGolden() end
    end

    if which == "normal" then
        collectNormal()
    elseif which == "golden" then
        collectGolden()
    end
end

(function()
    local controller = {
        Destroyed = false,
        AutoNormal = false,
        AutoGolden = false,
        AutoBeacon = false,
        AutoLobby = false,
        ActiveCollection = nil,
        CollectionTaskActive = false,
        SawGoldenGift = false,
        BeaconedForCurrentBatch = false,
        LobbyBusy = false,
        LastLobbyAttempt = 0
    }

    celestialIntegration.AutoFarmController = controller

    function controller:ResetCandidates()
        table.clear(availableNormalGifts)
        table.clear(availableGoldenGifts)
        scanIndex = 1
        scanIndexTwo = 1
    end

    function controller:CompactCandidates(candidateList)
        local seenGifts = {}

        for candidateIndex = #candidateList, 1, -1 do
            local gift = candidateList[candidateIndex]
            local validGift = gift
                and gift.Parent
                and gift.Transparency ~= 1
                and gift:FindFirstChild("Collect", true)

            if not validGift or seenGifts[gift] then
                table.remove(candidateList, candidateIndex)
            else
                seenGifts[gift] = true
            end
        end
    end

    function controller:TeleportToBeacon()
        local beacon = workspace:FindFirstChild("Beacon")
        local character = getChar(plr)
        local root, hitbox = getRoot(character)
        if not beacon or not root then
            return false
        end

        local beaconPosition = beacon:IsA("Model") and beacon:GetPivot().Position or beacon.Position
        local destination = CFrame.new(beaconPosition + Vector3.new(0, 4, 0))
        root.CFrame = destination

        if hitbox then
            hitbox.CFrame = destination
        end

        return true
    end

    function controller:GetLobbyOptionCFrame(option)
        return option:IsA("BasePart") and option.CFrame or option:GetPivot()
    end

    function controller:GetSortedLobbyOptions(selectFolder)
        local options = {}

        for _, option in ipairs(selectFolder:GetChildren()) do
            local optionName = string.lower(option.Name)
            local isRerollOption = optionName:find("reroll", 1, true) ~= nil

            if not isRerollOption and option:FindFirstChildOfClass("ProximityPrompt", true) then
                table.insert(options, option)
            end
        end

        table.sort(options, function(leftOption, rightOption)
            local leftCFrame = self:GetLobbyOptionCFrame(leftOption)
            local rightCFrame = self:GetLobbyOptionCFrame(rightOption)
            local leftOffset = spawnPart.CFrame:ToObjectSpace(leftCFrame).Position.X
            local rightOffset = spawnPart.CFrame:ToObjectSpace(rightCFrame).Position.X
            return leftOffset < rightOffset
        end)

        return options
    end

    function controller:ActivateLobbyOption(option, root)
        local prompt = option and option:FindFirstChildOfClass("ProximityPrompt", true)
        if not prompt or type(fireproximityprompt) ~= "function" then
            return false
        end

        root.CFrame = self:GetLobbyOptionCFrame(option) + Vector3.new(0, 3, 0)
        task.wait(0.3)
        fireproximityprompt(prompt)
        task.wait(0.7)
        return true
    end

    function controller:RunLobbySelection()
        if self.LobbyBusy or self.Destroyed then
            return
        end

        self.LobbyBusy = true
        self.LastLobbyAttempt = tick()

        local selectionSucceeded, selectionError = pcall(function()
            local selectFolder = workspace:FindFirstChild("Select")
            local character = getChar(plr)
            local root = getRoot(character)
            if not selectFolder or not spawnPart or not root then
                return
            end

            local options = self:GetSortedLobbyOptions(selectFolder)
            if #options == 0 then
                return
            end

            self:ActivateLobbyOption(options[#options], root)
            options = self:GetSortedLobbyOptions(selectFolder)

            table.sort(options, function(leftOption, rightOption)
                local leftOffset = spawnPart.CFrame:ToObjectSpace(self:GetLobbyOptionCFrame(leftOption)).Position.X
                local rightOffset = spawnPart.CFrame:ToObjectSpace(self:GetLobbyOptionCFrame(rightOption)).Position.X
                return math.abs(leftOffset) < math.abs(rightOffset)
            end)

            if options[1] then
                self:ActivateLobbyOption(options[1], root)
            end

            task.wait(1.3)
        end)

        self.LobbyBusy = false
        if not selectionSucceeded then
            warn("[NULLGUI Auto-Farm] Lobby selection failed: " .. tostring(selectionError))
        end
    end

    function controller:StartCollection(collectionType)
        if self.CollectionTaskActive or tweening or self.Destroyed then
            return false
        end

        self.CollectionTaskActive = true
        self.ActiveCollection = collectionType

        task.spawn(function()
            local collectionSucceeded, collectionError = pcall(function()
                collect(collectionType, true)
            end)

            self.CollectionTaskActive = false
            self.ActiveCollection = nil

            if not collectionSucceeded then
                warn("[NULLGUI Auto-Farm] Collection failed: " .. tostring(collectionError))
                return
            end

            if collectionType == "golden"
                and self.AutoBeacon
                and self.SawGoldenGift
                and not self.BeaconedForCurrentBatch
                and not getClosestGift(availableGoldenGifts)
            then
                self.BeaconedForCurrentBatch = self:TeleportToBeacon()
