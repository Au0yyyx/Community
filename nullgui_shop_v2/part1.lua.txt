    function configurationManager:NotifyResult(success, message)
        notif(message, success and "Configuration" or "Configuration Error")
    end

    function configurationManager:EnsureStorage()
        if type(writefile) ~= "function" then
            return false, "Your executor is missing writefile."
        end

        if type(readfile) ~= "function" then
            return false, "Your executor is missing readfile."
        end

        if type(isfile) ~= "function" then
            return false, "Your executor is missing isfile."
        end

        if type(makefolder) ~= "function" then
            return false, "Your executor is missing makefolder."
        end

        if type(isfolder) ~= "function" then
            return false, "Your executor is missing isfolder."
        end

        local storageReady, storageError = pcall(function()
            if not isfolder(self.RootFolder) then
                makefolder(self.RootFolder)
            end

            if not isfolder(self.ConfigFolder) then
                makefolder(self.ConfigFolder)
            end
        end)

        if not storageReady then
            return false, "Could not create the config folder: " .. tostring(storageError)
        end

        return true
    end

    function configurationManager:NormalizeName(configName)
        local normalizedName = tostring(configName or "")
            :gsub("^%s+", "")
            :gsub("%s+$", "")

        if normalizedName == "" then
            return nil, "Enter a config name first."
        end

        if #normalizedName > 48 then
            return nil, "Config names must be 48 characters or fewer."
        end

        if normalizedName:find("[^%w _%-]") then
            return nil, "Use only letters, numbers, spaces, underscores, or hyphens."
        end

        return normalizedName
    end

    function configurationManager:GetConfigPath(configName)
        return self.ConfigFolder .. "/" .. configName .. ".json"
    end

    function configurationManager:FindExistingName(configName)
        local lowercaseName = configName:lower()

        for _, existingName in ipairs(self.Manifest.Configs) do
            if existingName:lower() == lowercaseName then
                return existingName
            end
        end

        return nil
    end

    function configurationManager:SortConfigNames()
        table.sort(self.Manifest.Configs, function(leftName, rightName)
            return leftName:lower() < rightName:lower()
        end)
    end

    function configurationManager:WriteJson(path, value)
        local encodedSuccessfully, encodedValue = pcall(function()
            return HttpService:JSONEncode(value)
        end)

        if not encodedSuccessfully then
            return false, "Could not encode config data: " .. tostring(encodedValue)
        end

        local writtenSuccessfully, writeError = pcall(function()
            writefile(path, encodedValue)
        end)

        if not writtenSuccessfully then
            return false, "Could not write config data: " .. tostring(writeError)
        end

        return true
    end

    function configurationManager:ReadJson(path)
        local readSuccessfully, fileContents = pcall(function()
            return readfile(path)
        end)

        if not readSuccessfully then
            return nil, "Could not read config data: " .. tostring(fileContents)
        end

        local decodedSuccessfully, decodedValue = pcall(function()
            return HttpService:JSONDecode(fileContents)
        end)

        if not decodedSuccessfully then
            return nil, "Config data is invalid: " .. tostring(decodedValue)
        end

        return decodedValue
    end

    function configurationManager:SaveManifest()
        local storageReady, storageError = self:EnsureStorage()
        if not storageReady then
            return false, storageError
        end

        self.Manifest.Version = 1
        self:SortConfigNames()
        return self:WriteJson(self.ManifestPath, self.Manifest)
    end

    function configurationManager:LoadManifest()
        local storageReady, storageError = self:EnsureStorage()
        if not storageReady then
            return false, storageError
        end

        local manifest = {
            Version = 1,
            Configs = {},
            Autoload = nil
        }

        if isfile(self.ManifestPath) then
            local storedManifest, manifestError = self:ReadJson(self.ManifestPath)
            if not storedManifest then
                return false, manifestError
            end

            if type(storedManifest) ~= "table" then
                return false, "The config index is invalid."
            end

            if type(storedManifest.Configs) == "table" then
                manifest.Configs = storedManifest.Configs
            end

            if type(storedManifest.Autoload) == "string" then
                manifest.Autoload = storedManifest.Autoload
            end
        end

        local discoveredNames = {}
        for _, configName in ipairs(manifest.Configs) do
            local normalizedName = type(configName) == "string" and self:NormalizeName(configName) or nil
            if normalizedName and isfile(self:GetConfigPath(normalizedName)) then
                discoveredNames[normalizedName:lower()] = normalizedName
            end
        end

        if type(listfiles) == "function" then
            local listedSuccessfully, listedFiles = pcall(function()
                return listfiles(self.ConfigFolder)
            end)

            if listedSuccessfully and type(listedFiles) == "table" then
                for _, configPath in ipairs(listedFiles) do
                    local configName = tostring(configPath):match("([^/\\]+)%.json$")
                    local normalizedName = configName and self:NormalizeName(configName) or nil
                    if normalizedName then
                        discoveredNames[normalizedName:lower()] = normalizedName
                    end
                end
            end
        end

        manifest.Configs = {}
        for _, configName in pairs(discoveredNames) do
            table.insert(manifest.Configs, configName)
        end

        if manifest.Autoload and not discoveredNames[manifest.Autoload:lower()] then
            manifest.Autoload = nil
        end

        self.Manifest = manifest
        self:SortConfigNames()
        return self:SaveManifest()
    end

    function configurationManager:PackValue(value)
        local valueType = typeof(value)

        if valueType == "Color3" then
            return {
                NULLGUIType = "Color3",
                R = value.R,
                G = value.G,
                B = value.B
            }
        end

        if valueType == "EnumItem" then
            return {
                NULLGUIType = "EnumItem",
                Name = value.Name
            }
        end

        if valueType == "table" then
            local packedTable = {}
            for key, nestedValue in pairs(value) do
                packedTable[key] = self:PackValue(nestedValue)
            end
            return packedTable
        end

        if valueType == "boolean" or valueType == "number" or valueType == "string" then
            return value
        end

        return nil
    end

    function configurationManager:UnpackValue(value)
        if type(value) ~= "table" then
            return value
        end

        if value.NULLGUIType == "Color3" then
            return Color3.new(value.R or 0, value.G or 0, value.B or 0)
        end

        if value.NULLGUIType == "EnumItem" then
            return value.Name
        end

        local unpackedTable = {}
        for key, nestedValue in pairs(value) do
            unpackedTable[key] = self:UnpackValue(nestedValue)
        end
        return unpackedTable
    end

    function configurationManager:GetElementValue(elementRecord)
        local element = elementRecord.Element

        if elementRecord.MethodName == "CreateKeybind" then
            return element.CurrentKeybind
        end

        if elementRecord.MethodName == "CreateDropdown" then
            return element.CurrentOption
        end

        if elementRecord.MethodName == "CreateColorPicker" then
            return element.Color
        end

        return element.CurrentValue
    end

    function configurationManager:CaptureValues()
        local capturedValues = {}

        for flag, elementRecord in pairs(self.ElementsByFlag) do
            local currentValue = self:GetElementValue(elementRecord)
            if currentValue ~= nil then
                capturedValues[flag] = self:PackValue(currentValue)
            end
        end

        return capturedValues
    end

    function configurationManager:SaveConfig(configName, overwriteExisting)
        local storageReady, storageError = self:EnsureStorage()
        if not storageReady then
            return false, storageError
        end

        local normalizedName, nameError = self:NormalizeName(configName)
        if not normalizedName then
            return false, nameError
        end

        local existingName = self:FindExistingName(normalizedName)
        if existingName and not overwriteExisting then
            return false, "That config already exists. Select it and use overwrite."
        end

        if overwriteExisting and not existingName then
            return false, "Select an existing config to overwrite."
        end

        local savedName = existingName or normalizedName
        local configData = {
            Version = 1,
            Name = savedName,
            SavedAt = os.time(),
            Values = self:CaptureValues()
        }
        local configSaved, configError = self:WriteJson(self:GetConfigPath(savedName), configData)
        if not configSaved then
            return false, configError
        end

        if not existingName then
            table.insert(self.Manifest.Configs, savedName)
        end

        local manifestSaved, manifestError = self:SaveManifest()
        if not manifestSaved then
            return false, manifestError
        end

        self:RefreshDropdown(savedName)
        return true, overwriteExisting and "Config overwritten: " .. savedName or "Config saved: " .. savedName
    end

    function configurationManager:LoadConfig(configName, silent)
        local existingName = configName and self:FindExistingName(configName) or nil
        if not existingName then
            return false, "Select a saved config first."
        end

        local configData, configError = self:ReadJson(self:GetConfigPath(existingName))
        if not configData then
            return false, configError
        end

        if type(configData.Values) ~= "table" then
            return false, "That config does not contain valid settings."
        end

        local elementRecords = {}
        for flag, packedValue in pairs(configData.Values) do
            local elementRecord = self.ElementsByFlag[flag]
            if elementRecord then
                table.insert(elementRecords, {
                    Flag = flag,
                    PackedValue = packedValue,
                    ElementRecord = elementRecord
                })
            end
        end

        table.sort(elementRecords, function(leftRecord, rightRecord)
            local priorities = {
                CreateKeybind = 1,
                CreateInput = 2,
                CreateDropdown = 2,
                CreateSlider = 2,
                CreateColorPicker = 2,
                CreateToggle = 3
            }
            local leftPriority = priorities[leftRecord.ElementRecord.MethodName] or 2
            local rightPriority = priorities[rightRecord.ElementRecord.MethodName] or 2

            if leftPriority == rightPriority then
                return leftRecord.Flag < rightRecord.Flag
            end

            return leftPriority < rightPriority
        end)

        local failedSettings = 0
        for _, loadRecord in ipairs(elementRecords) do
            local settingLoaded, settingError = pcall(function()
                loadRecord.ElementRecord.Element:Set(self:UnpackValue(loadRecord.PackedValue))
            end)

            if not settingLoaded then
                failedSettings += 1
                warn("[NULLGUI Configuration] Could not load " .. loadRecord.Flag .. ": " .. tostring(settingError))
            end
        end

        self.SelectedName = existingName
        self:RefreshDropdown(existingName)

        if failedSettings > 0 then
            return true, "Loaded " .. existingName .. " with " .. failedSettings .. " setting error(s)."
        end

        if silent then
            return true
        end

        return true, "Config loaded: " .. existingName
    end

    function configurationManager:DeleteConfig(configName)
        local existingName = configName and self:FindExistingName(configName) or nil
        if not existingName then
            return false, "Select a saved config first."
        end

        if type(delfile) ~= "function" then
            return false, "Your executor is missing delfile."
        end

        local deletedSuccessfully, deleteError = pcall(function()
            delfile(self:GetConfigPath(existingName))
        end)

        if not deletedSuccessfully then
            return false, "Could not delete the config: " .. tostring(deleteError)
        end

        for configIndex, storedName in ipairs(self.Manifest.Configs) do
            if storedName == existingName then
                table.remove(self.Manifest.Configs, configIndex)
                break
            end
        end

        if self.Manifest.Autoload == existingName then
            self.Manifest.Autoload = nil
        end

        self.SelectedName = nil
        local manifestSaved, manifestError = self:SaveManifest()
        if not manifestSaved then
            return false, manifestError
        end

        self:RefreshDropdown()
        self:UpdateAutoloadLabel()
        return true, "Config deleted: " .. existingName
    end

    function configurationManager:SetAutoload(configName)
        local existingName = configName and self:FindExistingName(configName) or nil
        if not existingName then
            return false, "Select a saved config first."
        end

        self.Manifest.Autoload = existingName
        local manifestSaved, manifestError = self:SaveManifest()
        if not manifestSaved then
            return false, manifestError
        end

        self:UpdateAutoloadLabel()
        return true, "Autoload set to: " .. existingName
    end

    function configurationManager:ClearAutoload()
        self.Manifest.Autoload = nil
        local manifestSaved, manifestError = self:SaveManifest()
        if not manifestSaved then
            return false, manifestError
        end

        self:UpdateAutoloadLabel()
        return true, "Autoload cleared."
    end

    function configurationManager:Initialize()
        local manifestLoaded, manifestError = self:RefreshList()
        if not manifestLoaded then
            warn("[NULLGUI Configuration] " .. tostring(manifestError))
            if self.AutoloadLabel then
                self.AutoloadLabel:Set("Autoload: Filesystem unavailable")
            end
            return
        end

        if self.Manifest.Autoload then
            local configLoaded, configError = self:LoadConfig(self.Manifest.Autoload, true)
            if not configLoaded then
                warn("[NULLGUI Configuration] " .. tostring(configError))
            end
        end
    end

    local function sanitizeFlagPart(value)
        local sanitizedValue = tostring(value or "Control")
            :gsub("[^%w_]", "_")
            :gsub("_+", "_")
            :gsub("^_+", "")
            :gsub("_+$", "")

        if sanitizedValue == "" then
            return "Control"
        end

        return sanitizedValue
    end

    local function generateFlag(tabName, methodName, controlName)
        local baseFlag = table.concat({
            "NULLGUI",
            sanitizeFlagPart(tabName),
            sanitizeFlagPart(methodName),
            sanitizeFlagPart(controlName)
        }, "_")
        local occurrence = (flagOccurrences[baseFlag] or 0) + 1
        local generatedFlag = occurrence == 1 and baseFlag or baseFlag .. "_" .. occurrence

        while usedFlags[generatedFlag] do
            occurrence += 1
            generatedFlag = baseFlag .. "_" .. occurrence
        end

        flagOccurrences[baseFlag] = occurrence
        usedFlags[generatedFlag] = true
        return generatedFlag
    end

    local function installConfigurableMethod(tabName, tab, methodName)
        local originalMethod = tab[methodName]

        if type(originalMethod) ~= "function" then
            return
        end

        tab[methodName] = function(self, settings)
            local excludeFromConfig = settings and settings.NULLGUIExcludeFromConfig == true
            local isTransientDropdown = methodName == "CreateDropdown"
                and type(settings) == "table"
                and type(settings.Options) == "table"
                and #settings.Options == 0

            if type(settings) == "table" then
                if not excludeFromConfig and not isTransientDropdown and (settings.Flag == nil or settings.Flag == "") then
                    settings.Flag = generateFlag(tabName, methodName, settings.Name)
                elseif settings.Flag and settings.Flag ~= "" then
                    usedFlags[settings.Flag] = true
                end
            end

            local createdElement = originalMethod(self, settings)
            if not excludeFromConfig
                and not isTransientDropdown
                and type(settings) == "table"
                and settings.Flag
                and createdElement
            then
                configurationManager.ElementsByFlag[settings.Flag] = {
                    Element = createdElement,
                    MethodName = methodName,
                    Name = settings.Name
                }
            end

            return createdElement
        end
    end

    for tabName, tab in pairs(tabsByName) do
        for _, methodName in ipairs(configurableMethods) do
            installConfigurableMethod(tabName, tab, methodName)
        end
    end
