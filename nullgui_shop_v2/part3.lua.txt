            end)
        end

        safeCall(function()
            if not userInputService then
                return
            end

            local mousePosition = userInputService:GetMouseLocation()
            for buttonIndex = 0, 2 do
                virtualInputManager:SendMouseButtonEvent(
                    mousePosition.X,
                    mousePosition.Y,
                    buttonIndex,
                    false,
                    game,
                    0
                )
            end
        end)
    end

    function AutoStopController:ReleaseAbility(cancelMethods)
        self:ReleaseKeys()
        safeCall(self.Dependencies.ReleaseAbility, cancelMethods)
        safeCall(self.Dependencies.ReleaseMovementInput)

        local humanoid = self:GetParts()
        if humanoid then
            safeCall(function()
                humanoid:SetAttribute("UsingAbility", false)
                humanoid:SetAttribute("Sliding", false)
                humanoid:SetAttribute("PlayerMoveDirection", Vector3.zero)
            end)
        end
    end

    function AutoStopController:Capture()
        if self.Snapshot then
            return
        end

        local controls = self:GetControls()
        self.Snapshot = {
            Controls = controls,
            ControlsEnabled = readProperty(controls, "controlsEnabled"),
        }
    end

    function AutoStopController:Apply(humanoid)
        self:Capture()
        if not self.AbilityReleased then
            self:ReleaseAbility()
            self.AbilityReleased = true
        end

        local controls = self:GetControls()
        safeCall(function()
            controls:Disable()
        end)
        safeCall(self.Dependencies.ReleaseMovementInput)

        humanoid = humanoid or self:GetParts()
        if humanoid then
            safeCall(function()
                humanoid:SetAttribute("PlayerMoveDirection", Vector3.zero)
                humanoid:SetAttribute("UsingAbility", false)
                humanoid:SetAttribute("Sliding", false)
            end)
        end

        return true
    end

    function AutoStopController:Restore()
        local snapshot = self.Snapshot
        self.Snapshot = nil
        self.Active = false
        self.AbilityReleased = false

        local runService = self.Dependencies.RunService
        if runService then
            runService:UnbindFromRenderStep(self.InputNeutralBindingName)
        end

        if not snapshot then
            self:ReleaseKeys()
            return
        end

        if snapshot.Controls then
            if snapshot.ControlsEnabled == false then
                safeCall(function()
                    snapshot.Controls:Disable()
                end)
            else
                safeCall(function()
                    snapshot.Controls:Enable()
                end)
            end
        end

        self:ReleaseKeys()
        safeCall(self.Dependencies.OnRestored)
    end

    function AutoStopController:Finish()
        self.Generation += 1
        self.Deadline = 0

        if self.HeartbeatConnection then
            safeCall(function()
                self.HeartbeatConnection:Disconnect()
            end)
            self.HeartbeatConnection = nil
        end

        self:Restore()
    end

    function AutoStopController:Cancel()
        self.ScheduleGeneration += 1
        self:Finish()
    end

    function AutoStopController:StopFor(duration, abilityAlreadyReleased)
        duration = math.max(tonumber(duration) or 0, 0)
        self.Generation += 1
        local generation = self.Generation
        self.Deadline = os.clock() + duration
        self.Active = true
        self.AbilityReleased = abilityAlreadyReleased == true

        local runService = self.Dependencies.RunService
        if runService then
            runService:UnbindFromRenderStep(self.InputNeutralBindingName)
            runService:BindToRenderStep(
                self.InputNeutralBindingName,
                Enum.RenderPriority.Last.Value + 100,
                function()
                    if not self.Active then
                        return
                    end

                    safeCall(self.Dependencies.ReleaseMovementInput)
                    local currentHumanoid = self:GetParts()
                    self:Apply(currentHumanoid)
                end
            )
        end

        if self.HeartbeatConnection then
            safeCall(function()
                self.HeartbeatConnection:Disconnect()
            end)
            self.HeartbeatConnection = nil
        end

        local humanoid = self:GetParts()
        self:Apply(humanoid)
        safeCall(self.Dependencies.OnStopped)

        if runService and duration > 0 then
            self.HeartbeatConnection = runService.Heartbeat:Connect(function()
                if generation ~= self.Generation or not self.Active then
                    return
                end

                if os.clock() >= self.Deadline then
                    self:Finish()
                    return
                end

                local currentHumanoid = self:GetParts()
                self:Apply(currentHumanoid)
            end)
        elseif duration <= 0 then
            self:Finish()
        end
    end

    function AutoStopController:WaitForAbilityRelease(scheduleGeneration)
        local runService = self.Dependencies.RunService
        if not runService then
            return scheduleGeneration == self.ScheduleGeneration
        end

        local deadline = os.clock() + abilityReleaseSettleTimeout
        repeat
            runService.Heartbeat:Wait()

            if scheduleGeneration ~= self.ScheduleGeneration
                or celestialIntegration.Controller.Destroyed then
                return false
            end

            safeCall(self.Dependencies.ReleaseMovementInput)
            local humanoid = self:GetParts()
            if not humanoid or humanoid:GetAttribute("UsingAbility") ~= true then
                return true
            end
        until os.clock() >= deadline

        return true
    end

    function AutoStopController:StopAfter(
        delay,
        duration,
        activation,
        getActivation,
        cancelMethods
    )
        delay = math.max(tonumber(delay) or 0, 0)
        duration = math.max(tonumber(duration) or 0, 0)
        local scheduleGeneration = self.ScheduleGeneration

        task.delay(delay, function()
            if scheduleGeneration ~= self.ScheduleGeneration
                or celestialIntegration.Controller.Destroyed then
                return
            end

            if type(getActivation) == "function" and activation ~= nil then
                local success, currentActivation = pcall(getActivation)
                if not success or currentActivation ~= activation then
                    return
                end
            end

            self:ReleaseAbility(cancelMethods)
            if not self:WaitForAbilityRelease(scheduleGeneration) then
                return
            end

            self:StopFor(duration, true)
        end)
    end

    function AutoStopController:Destroy()
        self:Cancel()
    end

                if playerModuleScript then

                        if foundControls then
                            controlModule = controls
                        end
                    end
                end
            end

            if type(controlModule) == "table" then
                controlModule.inputMoveVector = zero

                local activeController

                        controlModule.GetActiveController,

                    end

                    activeController.moveVector = zero
                    activeController.forwardValue = 0
                    activeController.backwardValue = 0
                    activeController.leftValue = 0
                    activeController.rightValue = 0
                    activeController.isJumping = false
                end

                if type(controlModule.moveFunction) == "function" then
                    pcall(controlModule.moveFunction, plr, zero, true)
                end
            end

            if movement then
                movement.specialHeld = false
                movement.altSpecialHeld = false
                movement.jumpPressed = false
                movement.sprintHeld = false
                movement.PlayerMoveDirection = zero
            end

                humanoid:SetAttribute("PlayerMoveDirection", zero)
                humanoid:SetAttribute("UsingAbility", false)
                humanoid:SetAttribute("Sliding", false)

local function triggerOperatorReaction()
    local safety = celestialIntegration.Controller.EnemySafety
    local currentTime = time()
    if celestialIntegration.Controller.Destroyed
        or not safety.AutoStopOperator
        or currentTime - enemyReactionState.LastOperatorTrigger < 0.25 then
        return
    end

    enemyReactionState.LastOperatorTrigger = currentTime
    safety.OperatorDetections += 1
    local movement = getAbilityController()
    local abilityIsActive = isAbilityControllerActive(movement)
    local configuredDelay = abilityIsActive
        and safety.OperatorAbilityDelay
        or safety.OperatorIdleDelay
    local sharkTailDelay = safety.OperatorCancelWithSharkTail
        and canCancelWithSharkTail(movement)
        and safety.OperatorSharkTailCancelDelay
        or 0
    local delay = clampReactionDelay(
        configuredDelay + sharkTailDelay,
        enemyAttackWindows.Operator,
        safety.OperatorStopDuration,
        abilityReleaseSettleTimeout
    )
    releaseAndStop(delay, safety.OperatorStopDuration, {
        SharkTail = safety.OperatorCancelWithSharkTail,
        Jump = safety.OperatorCancelWithJump,
    })
end

local function updateOperatorReaction()
    local startSound = getOperatorStartSound()
    if startSound ~= enemyReactionState.OperatorStartSound then
        if enemyReactionState.OperatorStartConnection then
            enemyReactionState.OperatorStartConnection:Disconnect()
            enemyReactionState.OperatorStartConnection = nil
        end

        enemyReactionState.OperatorStartSound = startSound
        if startSound and startSound:IsA("Sound") then
            local startConnection = startSound.Played:Connect(function()
                task.defer(function()
                    if startSound.Parent then
                        triggerOperatorReaction()
                    end
                end)
            end)
            enemyReactionState.OperatorStartConnection = startConnection
            table.insert(celestialIntegration.Controller.Connections, startConnection)
        end
    end

    local operatorIsActive = shared.OperatorActive == true
    if operatorIsActive and not enemyReactionState.OperatorWasActive then
        triggerOperatorReaction()
    end
    enemyReactionState.OperatorWasActive = operatorIsActive
end

local function triggerMalfunctionReaction(alert)
    local safety = celestialIntegration.Controller.EnemySafety
    local currentTime = time()
    local isCombatAlert = alert
        and alert:IsA("Sound")
        and alert.Name == "VBAlert"

    if celestialIntegration.Controller.Destroyed
        or not safety.AutoStopMalfunction
        or not isCombatAlert
        or currentTime - enemyReactionState.LastMalfunctionTrigger < 0.025 then
        return
    end

    enemyReactionState.LastMalfunctionTrigger = currentTime
    safety.MalfunctionDetections += 1
    local movement = getAbilityController()
    local abilityIsActive = isAbilityControllerActive(movement)
    local configuredDelay = abilityIsActive
        and safety.MalfunctionAbilityDelay
        or safety.MalfunctionIdleDelay
    local sharkTailDelay = safety.MalfunctionCancelWithSharkTail
        and canCancelWithSharkTail(movement)
        and safety.MalfunctionSharkTailCancelDelay
        or 0
    local stopDuration = math.max(
        safety.MalfunctionStopDuration,
        minimumMalfunctionStopDuration
    )
    local delay = clampReactionDelay(
        configuredDelay + sharkTailDelay,
        enemyAttackWindows.Malfunction,
        stopDuration,
        abilityReleaseSettleTimeout
    )
    releaseAndStop(delay, stopDuration, {
        SharkTail = safety.MalfunctionCancelWithSharkTail,
        Jump = safety.MalfunctionCancelWithJump,
    })
end

local function updateMalfunctionReaction()
    local alert = getMalfunctionAlert()
    if alert ~= enemyReactionState.MalfunctionAlert then
        if enemyReactionState.MalfunctionAlertConnection then
            enemyReactionState.MalfunctionAlertConnection:Disconnect()
            enemyReactionState.MalfunctionAlertConnection = nil
        end

        enemyReactionState.MalfunctionAlert = alert
        enemyReactionState.MalfunctionAlertWasActive = false

        if alert and alert:IsA("Sound") then
            local alertConnection = alert.Played:Connect(function()
                task.defer(function()
                    triggerMalfunctionReaction(alert)
                end)
            end)
            enemyReactionState.MalfunctionAlertConnection = alertConnection
            table.insert(celestialIntegration.Controller.Connections, alertConnection)
        end
    end

    local alertIsActive = alert
        and alert:IsA("Sound")
        and alert.IsPlaying

    if alertIsActive and not enemyReactionState.MalfunctionAlertWasActive then
        triggerMalfunctionReaction(alert)
    end

    enemyReactionState.MalfunctionAlertWasActive = alertIsActive == true
end

local function triggerKolonaAbility()
    local safety = celestialIntegration.Controller.EnemySafety
    local humanoid = getCharacterControlParts()
    local abilityIsActive = humanoid and humanoid:GetAttribute("UsingAbility") == true

    if celestialIntegration.Controller.Destroyed
        or shared.KolonaActive ~= true
        or enemyReactionState.KolonaTriggered
        or not safety.AutoUseKolonaAbility
        or not humanoid
        or abilityIsActive then
        return
    end

    enemyReactionState.KolonaTriggered = true
    safety.KolonaDetections += 1
    local kolonaActivation = enemyReactionState.KolonaActivation
    local delay = clampReactionDelay(
        safety.KolonaAbilityDelay,
        enemyAttackWindows.Kolona,
        safety.KolonaAbilityDuration + 0.03
    )

    task.delay(delay, function()
        if celestialIntegration.Controller.Destroyed
            or shared.KolonaActive ~= true
            or enemyReactionState.KolonaActivation ~= kolonaActivation then
            return
        end

        local currentHumanoid = getCharacterControlParts()
        if not currentHumanoid
            or currentHumanoid:GetAttribute("UsingAbility") == true then
            return
        end

        currentHumanoid:SetAttribute("UsingAbility", true)
        enemyReactionState.ForcedAbilityHumanoid = currentHumanoid
        safety.FallbackAbilityCalls += 1
        beginAbility()

        local abilityHoldDuration = math.max(safety.KolonaAbilityDuration, 0.1)
        task.delay(abilityHoldDuration, releaseAbility)
    end)
end

local function getKolonaCheckWindow()
    local character = plr.Character
    if not character then
        return false
    end

    if enemyReactionState.KolonaWindowCharacter ~= character then
        enemyReactionState.KolonaWindowCharacter = character
        enemyReactionState.KolonaWindowFunction = nil
        enemyReactionState.KolonaWindowWasOpen = false
    end

    local windowFunction = enemyReactionState.KolonaWindowFunction
    local connectionGetter = getconnections or get_signal_cons
    local upvalueGetter = getupvalues or (debug and debug.getupvalues)

    if type(windowFunction) ~= "function"
        and type(connectionGetter) == "function"
        and debug
        and type(debug.info) == "function" then
        local foundConnections, signalConnections = pcall(
            connectionGetter,
            character:GetAttributeChangedSignal("UsingAbility")
        )

        if foundConnections then
            for _, signalConnection in ipairs(signalConnections) do
                local candidateFunction = signalConnection.Function
                if type(candidateFunction) == "function" then
                    local readSource, source = pcall(debug.info, candidateFunction, "s")
                    if readSource
                        and type(source) == "string"
                        and source:find("Kolona_AI", 1, true) then
                        windowFunction = candidateFunction
                        enemyReactionState.KolonaWindowFunction = candidateFunction
                        break
                    end
                end
            end
        end
    end

    if type(windowFunction) ~= "function" or type(upvalueGetter) ~= "function" then
        return false
    end

    local readUpvalues, upvalues = pcall(upvalueGetter, windowFunction)
    return readUpvalues and type(upvalues) == "table" and upvalues[1] == true
end

local function installKolonaTickWatchers()
    local numberLabel = getKolonaNumberLabel()
    local mainFrame = numberLabel and numberLabel.Parent
    local tickFolder = mainFrame and mainFrame:FindFirstChild("tick")
    if not tickFolder then
        return

    for _, tickSound in ipairs(tickFolder:GetChildren()) do
        if tickSound:IsA("Sound")
            and not enemyReactionState.KolonaTickConnections[tickSound] then
            local tickConnection = tickSound.Played:Connect(function()
                if getKolonaCheckWindow() then
                    triggerKolonaAbility()
                end
            end)
            enemyReactionState.KolonaTickConnections[tickSound] = tickConnection
            celestialIntegration.Controller.EnemySafety.KolonaTickWatchers += 1
            table.insert(celestialIntegration.Controller.Connections, tickConnection)
        end
    end
end

local function updateKolonaReaction()
    installKolonaTickWatchers()
    local kolonaIsActive = shared.KolonaActive == true
    local numberLabel = getKolonaNumberLabel()
    local currentNumber = numberLabel and tonumber(numberLabel.Text)
    local checkWindowIsOpen = getKolonaCheckWindow()

    if kolonaIsActive and not enemyReactionState.KolonaWasActive then
        enemyReactionState.KolonaActivation += 1
        enemyReactionState.KolonaTarget = currentNumber
        enemyReactionState.KolonaPreviousNumber = currentNumber
        enemyReactionState.KolonaCountStarted = false
        enemyReactionState.KolonaTriggered = false
        if currentNumber then
            connectKolonaTargetSound(currentNumber)
        end
    elseif kolonaIsActive and not enemyReactionState.KolonaTarget and currentNumber then
        enemyReactionState.KolonaTarget = currentNumber
        enemyReactionState.KolonaPreviousNumber = currentNumber
        connectKolonaTargetSound(currentNumber)
    end

    if checkWindowIsOpen and not enemyReactionState.KolonaWindowWasOpen then
        triggerKolonaAbility()
    end
    enemyReactionState.KolonaWindowWasOpen = checkWindowIsOpen

    if kolonaIsActive and currentNumber and enemyReactionState.KolonaTarget then
        if enemyReactionState.KolonaPreviousNumber ~= currentNumber then
            enemyReactionState.KolonaCountStarted = true
        end

        local reachedRequiredNumber = enemyReactionState.KolonaCountStarted
            and currentNumber == enemyReactionState.KolonaTarget
            and not enemyReactionState.KolonaTriggered

        if reachedRequiredNumber then
            triggerKolonaAbility()
        end

        enemyReactionState.KolonaPreviousNumber = currentNumber
    end

    if not kolonaIsActive and enemyReactionState.KolonaWasActive then
        enemyReactionState.KolonaActivation += 1
        enemyReactionState.KolonaTarget = nil
        enemyReactionState.KolonaPreviousNumber = nil
        enemyReactionState.KolonaCountStarted = false
        enemyReactionState.KolonaTriggered = false
        enemyReactionState.KolonaTargetSound = nil

        if enemyReactionState.KolonaTargetSoundConnection then
            enemyReactionState.KolonaTargetSoundConnection:Disconnect()
            enemyReactionState.KolonaTargetSoundConnection = nil
        end
    end

    enemyReactionState.KolonaWasActive = kolonaIsActive
end

local function getGiftCounterValue(counter, attributeName)
    local attributeValue = counter and counter:GetAttribute(attributeName)
    if typeof(attributeValue) == "number" then
        return math.max(math.floor(attributeValue), 0)
    end

    return 0
end

local function getGiftCounterPercentage(collected, maximum)
    if maximum <= 0 then
        return 0
    end

    return math.clamp(math.floor((collected / maximum) * 100 + 0.5), 0, 100)
end

local function updateCelestialGiftCounterLayout()
    if not gameGiftLabel or not normalGiftLabel or not goldenGiftLabel then
        return
    end

    local lineHeight = math.max(math.floor(gameGiftLabel.AbsoluteSize.Y * 0.75), 30)
    local labelHeight = math.max(math.floor(gameGiftLabel.AbsoluteSize.Y * 0.7), 28)
    local labelSize = UDim2.new(0.75, 0, 0, labelHeight)

    normalGiftLabel.AnchorPoint = Vector2.new(0.5, gameGiftLabel.AnchorPoint.Y)
    goldenGiftLabel.AnchorPoint = normalGiftLabel.AnchorPoint
    normalGiftLabel.Position = gameGiftLabel.Position + UDim2.fromOffset(0, lineHeight)
    goldenGiftLabel.Position = gameGiftLabel.Position + UDim2.fromOffset(0, lineHeight * 2)
    normalGiftLabel.Size = labelSize
    goldenGiftLabel.Size = labelSize
end

    local giftCounterGui = celestialIntegration.Controller.GiftCounterGui
    if not giftCounterGui or not gameGui or not gameGiftLabel then
        return
    end

    local roundIsActive = not inRound or inRound.Value == true
    giftCounterGui.Enabled = celestialIntegration.Controller.GiftCounterEnabled
        and not celestialIntegration.Controller.Destroyed
        and roundIsActive
        and gameGui.Enabled
        and gameGiftLabel.Visible

    local normalCounter = counters:FindFirstChild("Gift") or counters:FindFirstChild("Normal")
    local goldenCounter = counters:FindFirstChild("GoldenGift")
    local normalCollected = getGiftCounterValue(normalCounter, "Collected")
    local normalMaximum = getGiftCounterValue(normalCounter, "MaxGifts")
    local goldenCollected = getGiftCounterValue(goldenCounter, "Collected")
    local goldenMaximum = getGiftCounterValue(goldenCounter, "MaxGifts")
    local giftsUntilGolden = math.max(normalMaximum - normalCollected, 0)
    local normalPercentage = getGiftCounterPercentage(normalCollected, normalMaximum)
    local goldenPercentage = getGiftCounterPercentage(goldenCollected, goldenMaximum)

    normalGiftLabel.Text = string.format(
        "%d/%d (%d%%) %d gifts left",

        normalPercentage,

        goldenPercentage

    end

    normalGiftLabel = cloneGiftLabel("NormalGifts", Color3.new(1, 1, 1))
    goldenGiftLabel = cloneGiftLabel("GoldenGifts", Color3.new(1, 1, 0))

    local function track(connection)
        table.insert(celestialIntegration.Controller.Connections, connection)
    end

    track(gameGiftLabel:GetPropertyChangedSignal("Text"):Connect(updateCelestialGiftCounter))
    track(gameGiftLabel:GetPropertyChangedSignal("Position"):Connect(updateCelestialGiftCounterLayout))
    track(gameGiftLabel:GetPropertyChangedSignal("Size"):Connect(updateCelestialGiftCounterLayout))
    track(gameGiftLabel:GetPropertyChangedSignal("AbsoluteSize"):Connect(updateCelestialGiftCounterLayout))
    track(gameGiftLabel:GetPropertyChangedSignal("Visible"):Connect(updateCelestialGiftCounter))
    track(gameGui:GetPropertyChangedSignal("Enabled"):Connect(updateCelestialGiftCounter))

    if inRound then
        track(inRound.Changed:Connect(updateCelestialGiftCounter))
    end

    for _, counter in ipairs({ counters:FindFirstChild("Gift"), counters:FindFirstChild("GoldenGift") }) do
        if counter then
            track(counter.Changed:Connect(updateCelestialGiftCounter))
            track(counter:GetAttributeChangedSignal("Collected"):Connect(updateCelestialGiftCounter))
            track(counter:GetAttributeChangedSignal("MaxGifts"):Connect(updateCelestialGiftCounter))
        end
    end

    updateCelestialGiftCounterLayout()
    updateCelestialGiftCounter()
    return true
end

function celestialIntegration.Controller:SetGiftCounterEnabled(value)
    self.GiftCounterEnabled = value == true
    updateCelestialGiftCounter()
end

local function isShieldBreakMessage(value)
    if typeof(value) ~= "string" then
        return false
    end

    local lowerValue = string.lower(value)
    return lowerValue:find("shield lost to", 1, true) ~= nil
        or lowerValue:find("no shields remaining", 1, true) ~= nil
        or lowerValue:find("shield broken", 1, true) ~= nil
end

function celestialIntegration.Controller:HandleShieldNotification(...)
    if not self.AutoEnableOnShieldBreak or self.Enabled or self.Destroyed then
        return false
    end

    local arguments = table.pack(...)
    for argumentIndex = 1, arguments.n do
        if isShieldBreakMessage(arguments[argumentIndex]) then
            self.ShieldTriggers += 1
            self:SetEnabled(true)
            self:SetSafetyToggle(true)
            self:RefreshSafetyFilter()

            if notifOn then
                notif("Celestial safety enabled after shield break.", "Celestial")
            end

            return true
        end
    end

    return false
end

local function isEnemyUpdatePositionRemote(instance)
    return instance:IsA("RemoteEvent")
        and instance.Name == "UpdatePosition"
        and instance:IsDescendantOf(enemies)
end

local function installUpdatePositionReceiver(remote)
    if celestialIntegration.Controller.Destroyed or celestialIntegration.Controller.UpdatePositionReceivers[remote] then
        return false
    end

    local receiverConnection = remote.OnClientEvent:Connect(function()
    end)
    celestialIntegration.Controller.UpdatePositionReceivers[remote] = receiverConnection
    table.insert(celestialIntegration.Controller.Connections, receiverConnection)

    local cleanupConnection
    cleanupConnection = remote.Destroying:Connect(function()
        celestialIntegration.Controller.UpdatePositionReceivers[remote] = nil
        receiverConnection:Disconnect()
        cleanupConnection:Disconnect()
    end)
    table.insert(celestialIntegration.Controller.Connections, cleanupConnection)
    return true
end

local function isRealityBreakClientEvent(instance)
    if not instance:IsA("RemoteEvent") or instance.Name ~= "ClientEvent" then
        return false
    end

    local attackPart = instance.Parent
    if not attackPart or not attackPart:IsDescendantOf(enemies) then
        return false
    end

    local lowerAttackName = string.lower(attackPart.Name)
    return lowerAttackName:find("realitybreak", 1, true) ~= nil
        and attackPart:FindFirstChild("RealityBreak_ClientAI") ~= nil
end

local function restoreRealityBreakGuard(remote)
    local guardState = celestialIntegration.Controller.RealityBreakGuards[remote]
    if not guardState then
        return
    end

    celestialIntegration.Controller.RealityBreakGuards[remote] = nil

    if guardState.WrapperConnection then
        pcall(function()
            guardState.WrapperConnection:Disconnect()
        end)
    end

    if guardState.OriginalConnection and type(guardState.OriginalConnection.Enable) == "function" then
        pcall(function()
            guardState.OriginalConnection:Enable()
        end)
    end
end

function celestialIntegration.Controller:RestoreRealityBreakGuards()
    local guardedRemotes = {}
    for remote in pairs(self.RealityBreakGuards) do
        table.insert(guardedRemotes, remote)
    end

    for _, remote in ipairs(guardedRemotes) do
        restoreRealityBreakGuard(remote)
    end
end

local signalConnectionsFunction = getconnections or get_signal_cons

local function installRealityBreakGuard(remote)
    if celestialIntegration.Controller.Destroyed
        or celestialIntegration.Controller.RealityBreakGuards[remote]
        or type(signalConnectionsFunction) ~= "function" then
        return false
    end

    task.spawn(function()
        for _ = 1, 20 do
            if celestialIntegration.Controller.Destroyed or not remote.Parent then
                return
            end

            local foundConnections, signalConnections = pcall(signalConnectionsFunction, remote.OnClientEvent)
            if foundConnections then
                for _, originalConnection in ipairs(signalConnections) do
                    local readConnectionSuccessfully, originalFunction, disableFunction, enableFunction = pcall(
                        function()
                            return originalConnection.Function,
                                originalConnection.Disable,
                                originalConnection.Enable
                        end
                    )
                    local canDisable = readConnectionSuccessfully and type(disableFunction) == "function"
                    local canEnable = readConnectionSuccessfully and type(enableFunction) == "function"

                    if type(originalFunction) == "function" and canDisable and canEnable then
                        local disabledSuccessfully = pcall(function()
                            originalConnection:Disable()
                        end)

                        if disabledSuccessfully then
                            local wrapperConnection = remote.OnClientEvent:Connect(function(...)
                                if celestialIntegration.Controller.EnemySafety.AutoDisableRealityBreak then
                                    return
                                end

                                local arguments = table.pack(...)

                                if arguments[1] == "Pizza2" then
                                    local character = plr.Character
                                    local rootPart = character and character:FindFirstChild("HumanoidRootPart")
                                    if not rootPart then
                                        return
                                    end
                                end

                                task.spawn(function()
                                    local callbackSucceeded, callbackError = pcall(
                                        originalFunction,
                                        table.unpack(arguments, 1, arguments.n)
                                    )

                                    if not callbackSucceeded then
                                        warn("[NULLGUI RealityBreak Guard] "..tostring(callbackError))
                                    end
                                end)
                            end)

                            celestialIntegration.Controller.RealityBreakGuards[remote] = {
                                OriginalConnection = originalConnection,
                                WrapperConnection = wrapperConnection,
                            }

                            local cleanupConnection
                            cleanupConnection = remote.Destroying:Connect(function()
                                restoreRealityBreakGuard(remote)
                                cleanupConnection:Disconnect()
                            end)
                            table.insert(celestialIntegration.Controller.Connections, cleanupConnection)
                            return
                        end
                    end
                end
            end

            task.wait(0.05)
        end
    end)

    return true
end

local function installEnemyCompatibility(instance)
    if isEnemyUpdatePositionRemote(instance) then
        installUpdatePositionReceiver(instance)
    end

    if isRealityBreakClientEvent(instance) then
        installRealityBreakGuard(instance)
        return
    end

    if instance.Name == "RealityBreak_ClientAI" and instance.Parent then
        local clientEvent = instance.Parent:FindFirstChild("ClientEvent")
        if clientEvent and isRealityBreakClientEvent(clientEvent) then
            installRealityBreakGuard(clientEvent)
        end
    end
end

function celestialIntegration.Controller:Panic()
    if self.Destroyed then
        return
    end

    self.Destroyed = true
    self.AutoEnableOnShieldBreak = false
    self.Enabled = false
    self.EnemySafety.AutoStopOperator = false
    self.EnemySafety.AutoStopMalfunction = false
    self.EnemySafety.AutoUseKolonaAbility = false
    self.EnemySafety.AutoLookAwayVoidbreaker = false
    table.clear(self.EnemySafety.Friendly)
    RunService:UnbindFromRenderStep(self.LookAwayBindingName)
    self:ReleaseReactionAbility()
    self:RestoreReactionCharacter()
    self:RestoreRealityBreakGuards()
    self:DisconnectAll()
    self:DestroyGiftCounter()
    table.clear(self.UpdatePositionReceivers)
    celestialControllerRegistry[self] = nil

    if environment.CelestialVisibleHarmlessController == self then
        environment.CelestialVisibleHarmlessController = nil
    end

    if celestialNamecallDispatcher.ActiveController == self then
        celestialNamecallDispatcher.ActiveController = nil
        celestialNamecallDispatcher.Filter = nil
    end
end
