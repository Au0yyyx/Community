local function activateAltar(justTeleport)
    if activating and justTeleport == false then return end
    if justTeleport == nil then
        justTeleport = false
    end

    if not justTeleport then
        activating = true
    end

    if not selectedPrompt or not selectedPrompt.Parent then
        if notifOn then
            notif("Altar no longer exists.", "Not found")
        end
        activating = false
        return
    end

    local pPart = selectedPrompt.Parent
    local char = getChar(plr)
    local root, hitbox = getRoot(char)
    if not root or not hitbox then
        activating = false
        return
    end

    local prev = root.CFrame
    local pos = pPart.CFrame + pPart.CFrame.LookVector * -3
    Camera.CFrame = pos
    root.CFrame = pos
    hitbox.CFrame = pos

    local start = tick()
    repeat
        task.wait(.05)
        Camera.CFrame = pos
        root.CFrame = pos
        hitbox.CFrame = pos
    until (root.Position - pPart.Position).Magnitude < 6 or tick() - start >= 3

    if justTeleport then return end

    fireproximityprompt(selectedPrompt)

    task.wait(selectedPrompt.HoldDuration)

    root.CFrame = prev
    hitbox.CFrame = prev
    activating = false
end

    local keybindControl
    keybindControl = keyTab:CreateKeybind({
        Name = "Toggle "..keybindInfo.Label.." Auto Disable",
        CurrentKeybind = "Unknown",
        HoldToInteract = false,
        Flag = "NULLGUI"..keybindInfo.Enemy.."AutoDisableKeybind",
        Callback = function()
            if not serverDisablerKeybindsEnabled then
                return
            end

            local currentKeybind = keybindControl and keybindControl.CurrentKeybind
            if not currentKeybind or currentKeybind == "Unknown" then
                return
            end

            local toggle = serverDisablerToggles[keybindInfo.Enemy]
            if not toggle or type(toggle.Set) ~= "function" then
                return
            end

            local enabled = auto_disable[keybindInfo.Enemy] ~= true
            toggle:Set(enabled)

            if notifOn then
                notif(
                    keybindInfo.Label.." auto disable "..(enabled and "enabled." or "disabled."),
                    "Enemy Keybind"
                )
            end
        end
    })
end

celestialIntegration.MusicController = celestialIntegration.MusicController or {}
;(function()
    local musicController = celestialIntegration.MusicController
    local musicFolder = game:GetService("SoundService").MusicFolder

    function musicController:StopCurrentMusic()
        for _, sound in musicFolder:GetDescendants() do
            if sound:IsA("Sound") and sound.Playing then
                sound:Stop()
            end
        end

        local currentMusic = music.Value
        if currentMusic then
            currentMusic:Stop()
        end

        music.Value = nil
        self.CurrentCustom = nil
        self.CustomPlaying = false
    end

    function musicController:PlayMusic(sound)
        self:StopCurrentMusic()
        self.CustomPlaying = true
        sound:Play()
        music.Value = sound
        self.CurrentCustom = sound
    end

    musicTab:CreateButton({
        Name = "Stop Current Music",
        Callback = function()
            musicController:StopCurrentMusic()
        end
    })
    musicTab:CreateDivider()

    for _, soundOrFolder in musicFolder:GetChildren() do
        if soundOrFolder:IsA("Sound") then
            musicTab:CreateButton({
                Name = "Play "..soundOrFolder.Name,
                Callback = function()
                    musicController:PlayMusic(soundOrFolder)
                end
            })
        elseif soundOrFolder:IsA("Folder") and soundOrFolder.Name ~= "Intermission" then
            local displayName = soundOrFolder:FindFirstChild("Name")
            local normalSong = soundOrFolder.MainSong
            local collapseSong = soundOrFolder.EscapeSong

            musicTab:CreateSection(displayName.Value)
            musicTab:CreateButton({
                Name = "Play Normal",
                Callback = function()
                    musicController:PlayMusic(normalSong)
                end
            })
            musicTab:CreateButton({
                Name = "Play Collapse",
                Callback = function()
                    musicController:PlayMusic(collapseSong)
                end
            })
        end
    end
end)()
